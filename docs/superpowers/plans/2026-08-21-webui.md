# NeuralGate webui 管理后台前端 实现计划

> **面向 AI 代理的工作者:** 必需子技能:使用 superpowers:subagent-driven-development(推荐)或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框(`- [ ]`)语法来跟踪进度。

**目标:** 为 OSS 版提供 Vue3 SPA 管理后台前端,对接已有 admin API,以 go:embed 嵌入单二进制,闭环 OSS 部署体验。

**架构:** webui/ 前端源码(Vue3+Vite+TS+Element Plus)→ vite build 产物输出到 pkg/admin/webui/dist → pkg/admin/webui.go 用 go:embed 服务静态资源 + SPA fallback。开发用 vite dev proxy /api→:8081。

**技术栈:** Vue 3、Vite、TypeScript、Element Plus、vue-router、axios、gin(go:embed)

**规格:** `docs/superpowers/specs/2026-08-21-webui-design.md`

## 关键:后端 API 实际 JSON 字段名(前端类型必须精确对齐)

各 API 响应字段命名**不一致**(直接返回无 json tag 的结构体 vs 自定义 item):

| API | 字段命名 | 说明 |
|-----|----------|------|
| `GET /api/models` | 小写 `id,name,provider,provider_model,base_url,timeout,max_retries,retry_interval,weight,enabled,tags,created_at` | 自定义 item |
| `GET /api/api-keys` | 小写 `id,key_prefix,name,status,quota,used_quota,rate_limit,allowed_models,expires_at,created_at` | 自定义 item |
| `GET /api/models/:id/upstreams` | 小写 `id,base_url,weight,enabled,created_at` | 自定义 item |
| `GET /api/audit-logs` | **大写** `ID,RequestID,TenantID,ModelName,ResponseStatus,TotalTokens,Duration,IsStream,CreatedAt,...` | 直接返回 plugin.AuditLog(无 json tag) |
| `GET /api/audit-logs/:id` | 小写 `id,request_id,model_name,response_status,request_body,response_body,sse_chunks,reassembled,...` | gin.H |
| `GET /api/rate-limits` | **大写** `ID,TenantID,ModelName,RequestsPerSec,TokensPerMin,Strategy,Enabled,CreatedAt,UpdatedAt` | 直接返回 plugin.RateLimitConfig |
| `GET /api/system` | 小写 `version,build_time,git_commit,edition,uptime,db_status,audit_queue_status,rate_limiter_status` | gin.H |

SSE 分片 `plugin.SSEChunk` 无 json tag → 大写 `Index,Data,Timestamp,EventType`。

统一响应包装:`{code, message, data}`,成功 data 为上述结构;`code!==0` 或 HTTP 非 2xx 由拦截器报错。

`webui/src/api/client.ts`:

```ts
import axios from 'axios'
import { ElMessage } from 'element-plus'

export const client = axios.create({
  baseURL: '/api',
  timeout: 15000
})

// 响应拦截:统一错误处理 + 解包 data
client.interceptors.response.use(
  (resp) => {
    const body = resp.data
    if (body && typeof body === 'object' && 'code' in body && body.code !== 0) {
      ElMessage.error(body.message || '操作失败')
      return Promise.reject(new Error(body.message))
    }
    return resp
  },
  (err) => {
    const msg =
      err.response?.data?.message ||
      err.response?.data?.error?.message ||
      (err.code === 'ECONNABORTED' ? '请求超时' : '无法连接管理后台')
    ElMessage.error(msg)
    return Promise.reject(err)
  }
)

// 解包:取 data 字段
export async function unwrap<T>(p: Promise<{ data: { data: T } }>): Promise<T> {
  const resp = await p
  return resp.data.data
}
```

import { onMounted, reactive, ref } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import type { ModelItem, ModelCreateRequest, UpstreamItem, UpstreamRequest } from '../types'
import { listModels, createModel, updateModel, deleteModel, testModel, listUpstreams, createUpstream, updateUpstream, deleteUpstream } from '../api/model'

const models = ref<ModelItem[]>([])
const page = ref(1)
const size = ref(10)
const total = ref(0)
const loading = ref(false)
const saving = ref(false)
const modelDialog = ref(false)
const upstreamDialog = ref(false)
const editing = ref<ModelItem | null>(null)
const editingUpstream = ref<UpstreamItem | null>(null)
const currentModelForUpstream = ref<ModelItem | null>(null)

const modelForm = reactive<ModelCreateRequest>({
  name: '', provider: 'openai', provider_model: '', base_url: '', api_key: '',
  timeout: 60, max_retries: 2, weight: 1, enabled: true
})
const upstreamForm = reactive<UpstreamRequest>({ base_url: '', api_key: '', weight: 1, enabled: true })

async function load() {
  loading.value = true
  try {
    const data = await listModels(page.value, size.value)
    models.value = data.items
    total.value = data.total

    loading.value = false
  }

function openCreate() {
  editing.value = null
  Object.assign(modelForm, { name: '', provider: 'openai', provider_model: '', base_url: '', api_key: '', timeout: 60, max_retries: 2, weight: 1, enabled: true })
  modelDialog.value = true
}

function openEdit(row: ModelItem) {
  editing.value = row
  Object.assign(modelForm, { name: row.name, provider: row.provider, provider_model: row.provider_model, base_url: row.base_url, api_key: '', timeout: row.timeout, max_retries: row.max_retries, weight: row.weight, enabled: row.enabled })
  modelDialog.value = true
}

async function saveModel() {
  saving.value = true
  try {
    if (editing.value) {
      await updateModel(editing.value.id, modelForm)
    } else {
      await createModel(modelForm)
    }
    ElMessage.success('保存成功')
    modelDialog.value = false
    await load()
  } finally {
    saving.value = false
  }
}

async function removeModel(row: ModelItem) {

  await load()
}

  ElMessage.success(enabled ? '已启用' : '已禁用')
  await load()
}

  const res = await testModel(row.id)
  if (res.ok) {
    ElMessage.success(`连接正常，延迟 ${res.latency_ms}ms`)
  } else {
    ElMessage.error(`连接失败: ${res.error || '未知错误'}`)
  }
}

  if (expandedRows.includes(row)) {
    row._upstreams = await listUpstreams(row.id)
  }
}

function openUpstreamCreate(row: ModelItem) {
  currentModelForUpstream.value = row
  editingUpstream.value = null
  Object.assign(upstreamForm, { base_url: '', api_key: '', weight: 1, enabled: true })
  upstreamDialog.value = true

  upstreamDialog.value = true

    saving.value = false

  await deleteUpstream(up.id)
  ElMessage.success('已删除')
  row._upstreams = await listUpstreams(row.id)
}

onMounted(load)

package admin

import (
	"embed"
	"io/fs"
	"net/http"
	"strings"

	"github.com/gin-gonic/gin"
)

//go:embed all:webui/dist
var webuiFS embed.FS

// registerWebUI 注册静态资源与 SPA fallback
// 非 /api 且命中静态资源 → 直接服务;否则回退 index.html(SPA 前端路由)
func (s *AdminServer) registerWebUI(r *gin.Engine) {
	dist, err := fs.Sub(webuiFS, "webui/dist")
	if err != nil {
		return // embed 目录异常时不注册(不影响 API)
	}
	r.StaticFS("/assets", http.FS(dist))
	index := func(c *gin.Context) {
		data, _ := fs.ReadFile(dist, "index.html")
		c.Header("Content-Type", "text/html; charset=utf-8")
		_, _ = c.Writer.Write(data)
	}
	r.GET("/", index)
	r.NoRoute(func(c *gin.Context) {
		if strings.HasPrefix(c.Request.URL.Path, "/api/") {
			c.JSON(http.StatusNotFound, Response{Code: 404, Message: "not found"})
			return
		}
		index(c)
	})
}
